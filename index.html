☁️ إنشاء مشروع Supabase جديد
كل شركة تحتاج مشروع Supabase خاص بها.
اتبع الخطوات التالية لإنشاء مشروع مجاني جديد:
1
افتح موقع Supabase اذهب إلى supabase.com وسجّل دخولك بحساب GitHub أو Google أو البريد الإلكتروني
2
أنشئ مشروعاً جديداً اضغط على New Project ← اختر المنظمة ← اكتب اسم المشروع (مثل: atlas-company) ← أدخل كلمة سر قاعدة البيانات ← اختر المنطقة الأقرب
3
انتظر حتى يكتمل الإنشاء يستغرق عادةً 1-2 دقيقة حتى يظهر المشروع كـ Active ✅
4
أنشئ جدول stoc_items اذهب إلى SQL Editor وشغّل ← انسخ SQL الجاهز لإنشاء الجداول المطلوبة تلقائياً
📄 SQL الجاهز — انسخه وشغّله في SQL Editor 
-- جدول المنتجات
CREATE TABLE IF NOT EXISTS stoc_items (
  id          BIGSERIAL PRIMARY KEY,
  code        TEXT NOT NULL,
  name        TEXT NOT NULL,
  category    TEXT,
  price       NUMERIC(12,2) DEFAULT 0,
  stock       INT DEFAULT 0,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- جدول الحركات
CREATE TABLE IF NOT EXISTS stoc_transactions (
  id          BIGSERIAL PRIMARY KEY,
  item_code   TEXT NOT NULL,
  type        TEXT NOT NULL,
  qty         INT NOT NULL,
  note        TEXT,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);

-- تفعيل RLS
ALTER TABLE stoc_items        ENABLE ROW LEVEL SECURITY;
ALTER TABLE stoc_transactions ENABLE ROW LEVEL SECURITY;

-- سماح للقراءة والكتابة عبر anon key
CREATE POLICY "public_access_items" ON stoc_items
  FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "public_access_tx"    ON stoc_transactions
  FOR ALL USING (true) WITH CHECK (true);
