شغّل SQL التالي مرة واحدة في Supabase SQL Editor قبل البدء: 
CREATE TABLE IF NOT EXISTS stoc_licenses (
  id         BIGSERIAL PRIMARY KEY,
  company    TEXT NOT NULL,
  code       TEXT NOT NULL UNIQUE,
  sb_url     TEXT,
  sb_key     TEXT,
  expires    DATE,
  devices    INT DEFAULT 5,
  frozen     BOOLEAN DEFAULT FALSE,
  note       TEXT,
  sig        TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE stoc_licenses ENABLE ROW LEVEL SECURITY;

GRANT SELECT ON stoc_licenses TO anon;
GRANT SELECT ON stoc_licenses TO authenticated;

CREATE POLICY "anon_read_only" ON stoc_licenses
  FOR SELECT TO anon USING (true);
CREATE POLICY "admin_full_access" ON stoc_licenses
  FOR ALL TO authenticated USING (true) WITH CHECK (true);

📄 SQL الجاهز 
-- جدول المنتجات
CREATE TABLE IF NOT EXISTS stoc_items (
  id          BIGSERIAL PRIMARY KEY,
  code        TEXT NOT NULL,
  name        TEXT NOT NULL,
  category    TEXT,
  price       NUMERIC(12,2) DEFAULT 0,
  stock       INT DEFAULT 0,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
-- جدول الحركات
CREATE TABLE IF NOT EXISTS stoc_transactions (
  id          BIGSERIAL PRIMARY KEY,
  item_code   TEXT NOT NULL,
  type        TEXT NOT NULL,
  qty         INT NOT NULL,
  note        TEXT,
  created_at  TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE stoc_items        ENABLE ROW LEVEL SECURITY;
ALTER TABLE stoc_transactions ENABLE ROW LEVEL SECURITY;
CREATE POLICY "public_access_items" ON stoc_items    FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "public_access_tx"    ON stoc_transactions FOR ALL USING (true) WITH CHECK (true);
